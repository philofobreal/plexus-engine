import { isContinuousPlaybackStep, type CanvasVisualPresentation, type VisualAnalysisSnapshot } from '../types/CanvasVisualSource';
import { WORMHOLE_NATIVE_CAPACITY, WORMHOLE_NATIVE_WEAVE_CAPACITY, WORMHOLE_NATIVE_STRIDE, WORMHOLE_NATIVE_METRES_PER_UNIT, WORMHOLE_NATIVE_MATERIAL_STRIDE,
    WORMHOLE_NATIVE_COSMOS_CAPACITY, type WormholeNativeCosmosKind,
    type WormholeNativeFrame, type WormholeNativePoint, type WormholeNativeSink, type WormholeNativeSource as NativeSource } from '../types/WormholeNativeFrame';
import type { WormholeContextSeed } from '../types/WormholeContext';
import { cloneDefaultVisualTuning, tuningMorphDeltaSec } from '../config/visualTuning';
import { XR_WORMHOLE_MACROS, XR_WORMHOLE_BOOSTS } from '../config/xrWormholeTuning';
import { featureFlags } from '../config/featureFlags';
import { WormholeFrameContext } from './WormholeFrameContext';
import type { VisualRendererBackend } from './RendererBackend';
import { createWormholeSpatialFrame, resolveWormholeSpatialFrame } from './wormholeSpatialFrame';

/** Logical projection dimensions only. No surface, canvas, raster field or WebGL context. */
class GeometryBackend implements VisualRendererBackend {
    readonly width = 960; readonly height = 540; frameCount = 0;
    background() {} noStroke() {} noFill() {} fill() {} stroke() {} strokeWeight() {}
    line() {} circle() {} triangle() {} beginShape() {} vertex() {} endShape() {}
    radialGlow() {} radialDim() {} compositeRingTint() {} drawFieldRaster() {}
    beginFieldRaster(): null { return null; }
}
type MutableFrame = { -readonly [K in keyof WormholeNativeFrame]: WormholeNativeFrame[K] };
export class WormholeNativeSource implements NativeSource, WormholeNativeSink {
    private readonly context: WormholeFrameContext;
    private readonly backend = new GeometryBackend();
    private readonly weavePacket = { count: 0, data: new Float32Array(WORMHOLE_NATIVE_WEAVE_CAPACITY * WORMHOLE_NATIVE_STRIDE) };
    private readonly materialPacket = { amount: 0, detail: 0, bloom: 0,
        grains: new Float32Array(WORMHOLE_NATIVE_CAPACITY * WORMHOLE_NATIVE_MATERIAL_STRIDE),
        weave: new Float32Array(WORMHOLE_NATIVE_WEAVE_CAPACITY * WORMHOLE_NATIVE_MATERIAL_STRIDE) };
    private readonly packet: MutableFrame = { version: 1, generation: 0, serial: 0, songTime: 0, count: 0,
        data: new Float32Array(WORMHOLE_NATIVE_CAPACITY * WORMHOLE_NATIVE_STRIDE), weave: this.weavePacket, material: this.materialPacket };
    private readonly spatial = createWormholeSpatialFrame();
    private readonly cosmosPacket = { count: 0, data: new Float32Array(WORMHOLE_NATIVE_COSMOS_CAPACITY * WORMHOLE_NATIVE_STRIDE) };
    private ready = false; private shown = false; private disposed = false; private revision = 0;
    private previous: number | null = null; private lastPlaying = false; private lastWall = Number.NaN;
    private dirty = true; private snap = false; private interval = 1 / 36 - .004;
    private cost = 0;
    constructor(seed?: WormholeContextSeed) {
        this.packet.cosmos = this.cosmosPacket;
        this.context = new WormholeFrameContext(seed ?? { version: 1, rawTuning: cloneDefaultVisualTuning(),
            macros: XR_WORMHOLE_MACROS, boosts: XR_WORMHOLE_BOOSTS,
            policy: { gains: 'meta', semanticResolver: featureFlags.semanticResolver, semanticChoreography: featureFlags.semanticChoreography } });
        this.context.identity.setNativeSink(this);
    }
    get frame(): WormholeNativeFrame | null { return this.shown ? this.packet : null; }
    get focalPoint() { return this.context.identity.routeFocus; }
    get spatialFrame() { return this.shown ? this.spatial : null; }
    get lastRenderMs(): number { return this.cost; }
    async prepare(analysis: VisualAnalysisSnapshot | null): Promise<void> {
        if (this.disposed) return;
        const revision = ++this.revision;
        this.ready = false; this.shown = false; this.previous = null; this.packet.count = 0; this.weavePacket.count = 0; this.cosmosPacket.count = 0;
        this.context.prepare(analysis); this.context.identity.setNativeSink(this);
        if (analysis) {
            const presets = new Map<string, unknown>();
            await Promise.all([...new Set(analysis.performancePlan.points.map(p => p.preset))].map(async name => {
                const response = await fetch(`${import.meta.env.BASE_URL}visual-tuning-presets/${encodeURIComponent(name)}`);
                if (!response.ok) throw new Error(`Could not load Wormhole preset: ${name}`);
                presets.set(name, await response.json());
            }));
            if (this.disposed || revision !== this.revision) return;
            this.context.preparePlan(analysis.performancePlan, presets);
        }
        if (this.disposed || revision !== this.revision) return;
        this.packet.generation = revision; this.ready = true; this.dirty = true;
    }
    render(time: number, playing: boolean, continuousOverride?: boolean): boolean {
        if (!this.ready || this.disposed || !Number.isFinite(time) || time < 0) return false;
        const now = performance.now(), previous = this.previous, step = previous === null ? 0 : time - previous;
        const continuous = playing && this.lastPlaying && previous !== null && (continuousOverride ?? isContinuousPlaybackStep(step, (now - this.lastWall) / 1000));
        const jump = previous !== null && !continuous && (step < 0 || step > .25);
        if (previous !== null && !this.dirty && !jump && playing === this.lastPlaying && step < this.interval) return false;
        const dt = continuous ? Math.min(1, step) : tuningMorphDeltaSec(time, previous);
        this.context.step(time, playing, dt, previous, jump, previous === null || jump || this.snap);
        this.context.identity.setNativeSink(this);
        this.packet.count = 0; this.weavePacket.count = 0; this.cosmosPacket.count = 0; this.packet.songTime = time; this.packet.serial++;
        this.backend.frameCount++;
        this.context.identity.draw(this.backend, [], []);
        resolveWormholeSpatialFrame(this.spatial, time, this.context.state.visualTuning, this.context.identity.frameDrive, this.context.state);
        this.previous = time; this.lastPlaying = playing; this.lastWall = now;
        this.dirty = false; this.snap = false; this.shown = true; this.cost = performance.now() - now;
        return true;
    }
    setPresentation(presentation: CanvasVisualPresentation): void {
        const changed = this.context.setPresentation(presentation);
        this.dirty ||= changed.tuningChanged || changed.directionChanged; this.snap ||= changed.tuningChanged || changed.directionChanged;
        if (presentation.maxFrameRateHz && Number.isFinite(presentation.maxFrameRateHz)) this.interval = Math.max(0, 1 / presentation.maxFrameRateHz - .004);
    }
    beginMaterial(amount: number, detail: number, bloom: number): void {
        this.materialPacket.amount = amount; this.materialPacket.detail = detail; this.materialPacket.bloom = bloom;
    }
    cosmos(kind: WormholeNativeCosmosKind, id: number, head: WormholeNativePoint, tail: WormholeNativePoint,
        radius: number, red: number, green: number, blue: number, alpha: number): void {
        if(this.cosmosPacket.count >= WORMHOLE_NATIVE_COSMOS_CAPACITY)return;
        const index=this.cosmosPacket.count++, o=index*WORMHOLE_NATIVE_STRIDE, s=WORMHOLE_NATIVE_METRES_PER_UNIT;
        this.writeSegment(this.cosmosPacket.data,index,id,head,tail,radius,red,green,blue,alpha,0,id,0);
        // Cosmic discs are deliberately larger than grain widths, measured before projection.
        this.cosmosPacket.data[o+7]=Math.max(.001,radius*s);this.cosmosPacket.data[o+15]=kind;
    }
    grain(id: number, head: WormholeNativePoint, tail: WormholeNativePoint, radius: number,
        red: number, green: number, blue: number, alpha: number, generation: number, seed: number, depth: number, phase=0, energy=0): void {
        if (this.packet.count >= WORMHOLE_NATIVE_CAPACITY) return;
        this.writeMaterial(this.materialPacket.grains, this.packet.count, phase, energy, alpha);
        this.writeSegment(this.packet.data, this.packet.count++, id, head, tail, radius, red, green, blue, alpha, generation, seed, depth);
    }
    weave(id: number, head: WormholeNativePoint, tail: WormholeNativePoint, radius: number,
        red: number, green: number, blue: number, alpha: number, generation: number, seed: number, depth: number, phase=0, energy=0, materialAlpha=alpha): void {
        if (this.weavePacket.count >= WORMHOLE_NATIVE_WEAVE_CAPACITY) return;
        this.writeMaterial(this.materialPacket.weave, this.weavePacket.count, phase, energy, materialAlpha);
        this.writeSegment(this.weavePacket.data, this.weavePacket.count++, id, head, tail, radius, red, green, blue, alpha, generation, seed, depth);
    }
    private writeMaterial(data: Float32Array, index: number, phase: number, energy: number, alpha: number): void {
        const o=index*WORMHOLE_NATIVE_MATERIAL_STRIDE;
        data[o]=phase;data[o+1]=Math.min(1,Math.max(0,energy));data[o+2]=Math.min(1,Math.max(0,alpha/255));
    }
    private writeSegment(data: Float32Array, index: number, id: number, head: WormholeNativePoint, tail: WormholeNativePoint, radius: number,
        red: number, green: number, blue: number, alpha: number, generation: number, seed: number, depth: number): void {
        const o = index * WORMHOLE_NATIVE_STRIDE, s = WORMHOLE_NATIVE_METRES_PER_UNIT;
        data[o] = id;
        data[o + 1] = head.x * s; data[o + 2] = -head.y * s; data[o + 3] = -head.z * s;
        data[o + 4] = tail.x * s; data[o + 5] = -tail.y * s; data[o + 6] = -tail.z * s;
        data[o + 7] = Math.min(.08, Math.max(.004, radius * s));
        data[o + 8] = red / 255; data[o + 9] = green / 255; data[o + 10] = blue / 255;
        data[o + 11] = Math.min(1, Math.max(0, alpha / 255)); data[o + 12] = generation;
        data[o + 13] = seed; data[o + 14] = depth; data[o + 15] = 0;
    }
    dispose(): void {
        this.disposed = true; this.revision++; this.shown = false; this.ready = false;
        this.context.identity.setNativeSink(null); this.context.dispose(); this.packet.count = 0; this.weavePacket.count = 0; this.cosmosPacket.count=0;
    }
}
