import { isContinuousPlaybackStep, type CanvasVisualPresentation, type VisualAnalysisSnapshot } from '../types/CanvasVisualSource';
import type { WormholeContextSeed } from '../types/WormholeContext';
import type { WormholeNativeSource as NativeSource, WormholeNativeFrame } from '../types/WormholeNativeFrame';
import { NATIVE_WORKER_BYTES, NATIVE_COSMOS_BYTES, nativeWorkerViews, validNativeResponse, type NativeRenderRequest,
    type NativeFrameResponse, type NativeWorkerResponse } from '../types/WormholeNativeWorkerProtocol';
import { WormholeNativeSource } from './WormholeNativeSource';
import { createWormholeNativeWorker } from './createWormholeNativeWorker';

export interface NativeWorkerOptions {
    forceMain?:boolean; seed?:WormholeContextSeed;
    workerFactory?:()=>Worker; mainFactory?:(seed?:WormholeContextSeed)=>NativeSource;
    prepareTimeoutMs?:number; renderTimeoutMs?:number;
}
function createBank(){
    const views=nativeWorkerViews(new ArrayBuffer(NATIVE_WORKER_BYTES));
    return {frame:{version:1 as const,generation:0,serial:0,songTime:0,count:0,data:views.grains,
        weave:{count:0,data:views.weave},cosmos:{count:0,data:new Float32Array(NATIVE_COSMOS_BYTES/4)},material:{amount:0,detail:0,bloom:0,grains:views.grainMaterial,weave:views.weaveMaterial}},
        meta:null as NativeFrameResponse|null,latency:0};
}
type Desired={songTime:number;playing:boolean;epoch:number;revision:number};
const same=(a:Desired|null,b:Desired)=>!!a&&a.songTime===b.songTime&&a.playing===b.playing&&a.epoch===b.epoch&&a.revision===b.revision;

/** One request in flight, one latest desired time. No simulation, canvas or Three on this side. */
export class WormholeNativeWorkerSource implements NativeSource {
    private readonly options:NativeWorkerOptions;
    private worker:Worker|null=null;private local:NativeSource|null=null;
    private readonly emptyFocus={x:0,y:0};
    private readonly banks=[createBank(),createBank()];private shownBank=-1;private pendingBank=-1;
    private generation=0;private requestId=0;private epoch=0;private revision=0;private ready=false;private disposed=false;
    private desired:Desired|null=null;private lastSent:NativeRenderRequest|null=null;private lastSentAt=0;
    private observed:{time:number;playing:boolean;wall:number}|null=null;
    private flight:{request:NativeRenderRequest;sentAt:number}|null=null;private timer:ReturnType<typeof setTimeout>|null=null;
    private preparing:{generation:number;resolve:()=>void;reject:(error:Error)=>void}|null=null;
    private analysis:VisualAnalysisSnapshot|null=null;private presentation:CanvasVisualPresentation={};
    private reason='';private producerMs=0;private copyMs=0;
    onFrameReady:(()=>void)|null=null;onError:((message:string)=>void)|null=null;
    constructor(options:NativeWorkerOptions={}){this.options=options;}
    get executionLabel():string{return this.local?`main geometry (fallback: ${this.reason}); no background canvas`:'worker geometry; no background canvas';}
    get frame():WormholeNativeFrame|null{return this.local?.frame??(this.shownBank>=0?this.banks[this.shownBank].frame:null);}
    get focalPoint(){return this.local?.frame?this.local.focalPoint:this.shownBank>=0?this.banks[this.shownBank].meta!.focal:this.emptyFocus;}
    get spatialFrame(){return this.local?.frame?this.local.spatialFrame:this.shownBank>=0?this.banks[this.shownBank].meta!.spatial:null;}
    /** Producer CPU time (worker when active), not main-thread frame cost. */
    get lastRenderMs():number{return this.local?.lastRenderMs??this.producerMs;}
    get frameLatencyMs():number{return this.local?0:this.shownBank>=0?this.banks[this.shownBank].latency:0;}
    get lastTransferMs():number{return this.copyMs;}
    prepare(analysis:VisualAnalysisSnapshot|null):Promise<void>{
        if(this.disposed)return Promise.resolve();
        this.preparing?.resolve();this.preparing=null;this.stopWorker();this.local?.dispose();this.local=null;
        this.analysis=analysis;this.generation++;this.ready=false;this.shownBank=-1;this.pendingBank=-1;
        this.desired=null;this.lastSent=null;this.observed=null;this.epoch++;this.producerMs=0;
        return new Promise((resolve,reject)=>{
            this.preparing={generation:this.generation,resolve,reject};
            if(this.options.forceMain){this.fallback('xrBackgroundThread=main');return;}
            try {
                const worker=(this.options.workerFactory??createWormholeNativeWorker)();this.worker=worker;
                worker.onmessage=event=>this.receive(worker,event.data as NativeWorkerResponse);
                worker.onerror=event=>{event.preventDefault?.();if(worker===this.worker)this.fallback(event.message||'worker error');};
                worker.onmessageerror=()=>{if(worker===this.worker)this.fallback('worker message could not be decoded');};
                this.arm(this.options.prepareTimeoutMs??15000,'worker startup timed out');
                // Copy analysis/seed: the host retains the immutable publication; no PCM/analysis is detached.
                worker.postMessage({type:'prepare',version:1,generation:this.generation,analysis,seed:this.options.seed,presentation:this.presentation});
            }catch(error){this.fallback(error instanceof Error?error.message:String(error));}
        });
    }
    setPresentation(presentation:CanvasVisualPresentation):void{
        const next={...this.presentation,...presentation,macros:{...this.presentation.macros,...presentation.macros},
            grainMaterial:{...this.presentation.grainMaterial,...presentation.grainMaterial},cosmos:{...this.presentation.cosmos,...presentation.cosmos}} as CanvasVisualPresentation;
        if(JSON.stringify(next)===JSON.stringify(this.presentation))return;
        this.presentation=next;this.revision++;this.pendingBank=-1;this.desired=null;this.local?.setPresentation(next);
    }
    render(songTime:number,playing:boolean):boolean{
        if(this.disposed||!this.ready||!Number.isFinite(songTime)||songTime<0)return false;
        if(this.local)return this.local.render(songTime,playing);
        const wall=performance.now(),previous=this.observed;
        if(previous){const step=songTime-previous.time;
            if(playing!==previous.playing||step<0||(!playing&&step!==0)||(step>.25&&!isContinuousPlaybackStep(step,(wall-previous.wall)/1000))){this.epoch++;this.pendingBank=-1;}
        }
        this.observed={time:songTime,playing,wall};
        this.desired={songTime,playing,epoch:this.epoch,revision:this.revision};
        let changed=false;
        if(this.pendingBank>=0){const bank=this.banks[this.pendingBank],meta=bank.meta!;
            if(meta.epoch===this.epoch&&meta.revision===this.revision){this.shownBank=this.pendingBank;this.producerMs=meta.renderMs;changed=true;}
            this.pendingBank=-1;
        }
        this.flush();return changed;
    }
    private flush():void{
        if(!this.worker||!this.ready||this.flight||!this.desired||same(this.lastSent,this.desired))return;
        const desired=this.desired,now=performance.now(),last=this.lastSent;
        const request:NativeRenderRequest={type:'render',version:1,generation:this.generation,requestId:++this.requestId,
            ...desired,continuous:!!last&&last.playing&&desired.playing&&last.epoch===desired.epoch
                &&isContinuousPlaybackStep(desired.songTime-last.songTime,(now-this.lastSentAt)/1000),presentation:this.presentation};
        this.flight={request,sentAt:now};this.lastSent=request;this.lastSentAt=now;
        try{this.arm(this.options.renderTimeoutMs??5000,'native worker render timed out');this.worker.postMessage(request);}
        catch(error){this.fallback(error instanceof Error?error.message:String(error));}
    }
    private receive(worker:Worker,message:NativeWorkerResponse):void{
        if(this.disposed||worker!==this.worker)return;
        if(message.version!==1||message.generation!==this.generation)return;
        if(message.type==='error'){this.fallback(message.message);return;}
        if(message.type==='ready'){
            if(this.ready)return;
            if(message.canvasFree!==true){this.fallback('native worker capability mismatch');return;}
            this.clearTimer();this.ready=true;this.settle();this.onFrameReady?.();return;
        }
        const flight=this.flight;if(!flight||message.requestId!==flight.request.requestId)return;
        const request=flight.request;
        if(message.epoch!==request.epoch||message.revision!==request.revision||message.songTime!==request.songTime){this.fallback('native worker frame identity mismatch');return;}
        this.clearTimer();this.flight=null;
        try {
            if(message.type==='frame'){
                if(!validNativeResponse(message))throw new Error('Invalid native worker frame.');
                if(message.epoch===this.epoch&&message.revision===this.revision){
                    const start=performance.now(),index=this.shownBank===0?1:0,bank=this.banks[index];
                    const input=nativeWorkerViews(message.buffer),frame=bank.frame;
                    frame.data.set(input.grains.subarray(0,message.count*16));frame.weave.data.set(input.weave.subarray(0,message.weaveCount*16));
                    frame.material.grains.set(input.grainMaterial.subarray(0,message.count*3));frame.material.weave.set(input.weaveMaterial.subarray(0,message.weaveCount*3));
                    frame.generation=message.generation;frame.serial=message.serial;frame.songTime=message.songTime;frame.count=message.count;frame.weave.count=message.weaveCount;
                    frame.material.amount=message.amount;frame.material.detail=message.detail;frame.material.bloom=message.bloom;
                    frame.cosmos.count=message.cosmosCount??0;
                    if(message.cosmosBuffer)frame.cosmos.data.set(new Float32Array(message.cosmosBuffer,0,frame.cosmos.count*16));
                    bank.meta=message;bank.latency=performance.now()-flight.sentAt;this.pendingBank=index;this.copyMs=performance.now()-start;
                }
                // Return exactly once, before the next render. The presented bank is never transferred.
                worker.postMessage({type:'release',version:1,generation:this.generation,requestId:message.requestId,buffer:message.buffer,
                    ...(message.cosmosBuffer?{cosmosBuffer:message.cosmosBuffer}:{})},message.cosmosBuffer?[message.buffer,message.cosmosBuffer]:[message.buffer]);
            }
            this.flush();
        }catch(error){this.fallback(error instanceof Error?error.message:String(error));return;}
        if(this.pendingBank>=0)this.onFrameReady?.();
    }
    private settle(error?:Error):void{
        const preparation=this.preparing;if(!preparation||preparation.generation!==this.generation)return;
        this.preparing=null;if(error)preparation.reject(error);else preparation.resolve();
    }
    private fallback(reason:string):void{
        if(this.disposed||this.local)return;
        this.reason=reason.slice(0,160);this.stopWorker();this.ready=false;this.pendingBank=-1;this.desired=null;
        const generation=this.generation;
        try{
            const local=(this.options.mainFactory??(seed=>new WormholeNativeSource(seed)))(this.options.seed);this.local=local;
            local.setPresentation(this.presentation);
            void local.prepare(this.analysis).then(()=>{
                if(this.disposed||generation!==this.generation||local!==this.local)return;
                this.ready=true;this.settle();this.onFrameReady?.();
            },error=>{if(!this.disposed&&generation===this.generation){const e=error instanceof Error?error:new Error(String(error));this.settle(e);this.onError?.(e.message);}});
        }catch(error){const e=error instanceof Error?error:new Error(String(error));this.settle(e);this.onError?.(e.message);}
    }
    private arm(ms:number,reason:string):void{this.clearTimer();this.timer=setTimeout(()=>this.fallback(reason),ms);}
    private clearTimer():void{if(this.timer!==null){clearTimeout(this.timer);this.timer=null;}}
    private stopWorker():void{
        this.clearTimer();if(this.worker){this.worker.onmessage=null;this.worker.onerror=null;this.worker.onmessageerror=null;this.worker.terminate();this.worker=null;}this.flight=null;
    }
    dispose():void{
        if(this.disposed)return;this.disposed=true;this.ready=false;this.stopWorker();this.local?.dispose();this.local=null;
        this.preparing?.resolve();this.preparing=null;this.shownBank=-1;this.pendingBank=-1;this.desired=null;
        this.onFrameReady=null;this.onError=null;
    }
}
