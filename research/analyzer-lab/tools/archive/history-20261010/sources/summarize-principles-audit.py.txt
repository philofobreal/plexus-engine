def without_timings(value):
    if isinstance(value, dict):
        return {k: without_timings(v) for k, v in value.items()
                if k not in {'elapsedMs', 'curveMs', 'peakMs'}}
    if isinstance(value, list):
        return [without_timings(v) for v in value]
    return value
