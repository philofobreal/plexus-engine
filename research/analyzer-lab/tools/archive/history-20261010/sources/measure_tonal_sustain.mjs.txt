import fs from 'node:fs';
import { dirname, join, resolve } from 'node:path';
import vm from 'node:vm';
import ts from 'typescript';
let input = '';
for await (const chunk of process.stdin) input += chunk;
const request = JSON.parse(input);
const cache = new Map();
function load(path) {
  if(cache.has(path)) return cache.get(path).exports;
  const module = {exports:{}};cache.set(path,module);
  let source=fs.readFileSync(path,'utf8');
  if(request.baseline && path.endsWith('FeatureExtractor.ts'))
    source=fs.readFileSync('research/analyzer-lab/.cache/FeatureExtractor.algorithm5.ts','utf8');
  const code=ts.transpileModule(source,{compilerOptions:{module:ts.ModuleKind.CommonJS,target:ts.ScriptTarget.ES2022}}).outputText;
  vm.runInNewContext(code,{module,exports:module.exports,Float32Array,Math,Number,
    require: name=> {const base=resolve(dirname(path),name);return load(fs.existsSync(base+'.ts')?base+'.ts':join(base,'index.ts'));}});
  return module.exports;
}
const bytes=fs.readFileSync(request.pcmPath);
const samples=new Float32Array(bytes.buffer.slice(bytes.byteOffset,bytes.byteOffset+bytes.byteLength));
const result=load(resolve('src/analyzer/analyzeAudio.ts')).analyzeAudio({samples,sampleRate:request.sampleRate});
process.stdout.write(JSON.stringify({events:result.events,beats:result.beats,bpm:result.bpm,
 barStarts:result.barStarts,timingConfidence:result.timingConfidence,
 percussiveCues:result.trackAnalysis.cues.filter(c=>c.reasons?.includes('percussive-onset'))}));
