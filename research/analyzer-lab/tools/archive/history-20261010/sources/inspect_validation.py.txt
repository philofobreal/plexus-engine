import json
from pathlib import Path
root=Path(__file__).resolve().parents[1]
for name in ('drums-v1-on-v2','drums-v2-development'):

    print(json.dumps({'model':name,'ridge':model['ridge'],'validation':model['validationSelection'],
                    'selection':model.get('hyperparameterSelection')},indent=2))
